# ---- Prowlarr ----------------------------------------------------------------
# Internal URL the bridge uses to reach Prowlarr. In Docker this is the service
# name + internal port; on the host it'd be http://localhost:9696
PROWLARR_URL=http://prowlarr:9696
# Settings -> General -> Security -> API Key in Prowlarr
PROWLARR_API_KEY=changeme

# Optional: restrict to Prowlarr categories. XXX category group is 6000.
# Leave blank to search everything. Comma-separated, e.g. 6000
PROWLARR_CATEGORIES=

# ---- TorBox ------------------------------------------------------------------
# Only needed for paths that hit TorBox directly (usenet by default, and torrents
# if you set TORRENT_ADD_MODE=torbox). Get it from TorBox -> Settings -> API.
TORBOX_API_KEY=changeme

# ---- Add routing -------------------------------------------------------------
# torrent: "prowlarr" grabs via Prowlarr -> rdtclient -> TorBox (recommended,
#          reuses your existing pipeline). "torbox" pushes straight to TorBox.
TORRENT_ADD_MODE=prowlarr
# usenet: "torbox" pushes the NZB straight to TorBox's usenet API (recommended,
#         since you have no usenet download client). "prowlarr" grabs via
#         Prowlarr if you later add SABnzbd/NZBGet.
USENET_ADD_MODE=torbox
